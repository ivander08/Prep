// C++ execution harness.
//
// A real file rather than a string template, for the same reason as the Java harness:
// generating C++ from a TypeScript template means every backslash and quote must survive
// several escaping layers, which produced uncompilable code more than once.
//
// Three placeholders are substituted at run time (see harnesses.ts). They are described
// here rather than written literally, because a literal mention would itself be replaced.
//
// Headers MUST precede the user's code: a LeetCode C++ submission assumes the standard
// library is already included and does not include anything itself.

#include <bits/stdc++.h>
using namespace std;

__USER_CODE__

namespace mini {

static string b64decode(const string& in) {
  static const string chars = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";
  int T[256];
  for (int i = 0; i < 256; i++) T[i] = -1;
  for (int i = 0; i < 64; i++) T[(unsigned char)chars[i]] = i;
  string out;
  int val = 0, bits = -8;
  for (unsigned char c : in) {
    if (T[c] == -1) continue;
    val = (val << 6) + T[c];
    bits += 6;
    if (bits >= 0) { out.push_back(char((val >> bits) & 0xFF)); bits -= 8; }
  }
  return out;
}

struct Case { string args; string expected; string error; };

static string trim(const string& s) {
  size_t a = s.find_first_not_of(" \t\n\r");
  size_t b = s.find_last_not_of(" \t\n\r");
  return a == string::npos ? "" : s.substr(a, b - a + 1);
}

// Escape a string for embedding inside a JSON string literal.
//
// The harness builds its output line by concatenation, and a string argument arrives
// JSON-encoded, so it always contains quotes. Emitted raw, they terminated the JSON string
// early and the whole run was reported as "harness emitted non-JSON" instead of as a case
// result — a passing case and a crashing one looked identical. Control characters become
// spaces for the same reason they do in the Java harness: a raw control byte is not valid
// inside a JSON string.
//
// Character codes rather than backslash literals throughout: a literal must survive this
// file's own escaping layers, and that has broken before. Numbers have nothing to escape.
static string jsonEscape(const string& s) {
  string out;
  out.reserve(s.size() + 8);
  for (unsigned char c : s) {
    if (c == 92) { out += (char)92; out += (char)92; }
    else if (c == 34) { out += (char)92; out += (char)34; }
    else if (c < 32) out += (char)32;
    else out += (char)c;
  }
  return out;
}

// Decode the escape sequences inside a JSON string literal.
//
// A string argument arrives JSON-encoded, so a tab is still the two characters `\` `t` when
// it reaches the harness. Undecoded, the value differs from the caller's by its escapes:
// visible as a literal backslash in the output, and a silently wrong answer for a concept
// such as `normalizeSpaces`, which is entirely about whitespace. Python, JavaScript and Go
// all decode this for free; C++ is the only one that has to do it by hand.
static string jsonUnescape(const string& s) {
  string out;
  out.reserve(s.size());
  for (size_t i = 0; i < s.size(); i++) {
    if (s[i] != (char)92 || i + 1 >= s.size()) { out += s[i]; continue; }
    char e = s[++i];
    switch (e) {
      case 'n': out += (char)10; break;
      case 't': out += (char)9; break;
      case 'r': out += (char)13; break;
      case 'b': out += (char)8; break;
      case 'f': out += (char)12; break;
      case '/': out += (char)47; break;
      case '"': out += (char)34; break;
      case 92: out += (char)92; break;
      case 'u': {
        if (i + 4 < s.size()) {
          int cp = (int)strtol(s.substr(i + 1, 4).c_str(), nullptr, 16);
          i += 4;
          if (cp < 0x80) {
            out += (char)cp;
          } else if (cp < 0x800) {
            out += (char)(0xC0 | (cp >> 6));
            out += (char)(0x80 | (cp & 0x3F));
          } else {
            out += (char)(0xE0 | (cp >> 12));
            out += (char)(0x80 | ((cp >> 6) & 0x3F));
            out += (char)(0x80 | (cp & 0x3F));
          }
        }
        break;
      }
      default: out += e; break;
    }
  }
  return out;
}

// Strip the surrounding quotes from a JSON string literal, and decode its escapes.
//
// A scalar string argument is JSON-encoded in the payload, so it arrives as `"()"` and the
// quotes are part of the transport, not the value. Without this the parameter is two
// characters longer than the caller intended, and every string argument is wrong by two.
static string unquote(const string& s) {
  string t = trim(s);
  if (t.size() >= 2 && t.front() == '"' && t.back() == '"') return jsonUnescape(t.substr(1, t.size() - 2));
  return t;
}

// Split a bracketed list into top-level elements, respecting nesting and quotes.
static vector<string> splitTop(const string& s) {
  vector<string> out;
  int depth = 0;
  bool inStr = false;
  bool esc = false;
  size_t start = 0;
  for (size_t i = 0; i < s.size(); i++) {
    char c = s[i];
    // A backslash escapes the next character. Without this, `"a\"b"` toggled inStr twice and
    // the following comma was treated as string content, merging two arguments into one.
    if (esc) { esc = false; continue; }
    if (c == '\\') { esc = true; continue; }
    if (c == '"') inStr = !inStr;
    else if (!inStr && (c == '[' || c == '{')) depth++;
    else if (!inStr && (c == ']' || c == '}')) depth--;
    else if (!inStr && c == ',' && depth == 0) { out.push_back(s.substr(start, i - start)); start = i + 1; }
  }
  if (start < s.size()) out.push_back(s.substr(start));
  return out;
}

static int matchBracket(const string& s, int from) {
  int d = 0;
  for (int i = from; i < (int)s.size(); i++) {
    if (s[i] == '[') d++;
    else if (s[i] == ']') { if (--d == 0) return i + 1; }
  }
  return (int)s.size();
}

static string parseFn(const string& p) {
  auto k = p.find("fn");
  if (k == string::npos) return "";
  auto c = p.find(':', k);
  auto q1 = p.find('"', c);
  auto q2 = p.find('"', q1 + 1);
  return p.substr(q1 + 1, q2 - q1 - 1);
}

static vector<Case> parseCases(const string& p) {
  vector<Case> out;
  auto k = p.find("cases");
  if (k == string::npos) return out;
  auto arr = p.find('[', k);
  if (arr == string::npos) return out;
  int i = (int)arr + 1, depth = 0, start = -1;
  for (int j = i; j < (int)p.size(); j++) {
    if (p[j] == '{') { if (depth == 0) start = j; depth++; }
    else if (p[j] == '}') {
      depth--;
      if (depth == 0) {
        string obj = p.substr(start, j - start + 1);
        // A case whose `args` or `expected` field is missing used to fall through to
        // `obj.substr(npos, …)`, which throws out_of_range from OUTSIDE the per-case try — so the
        // whole run died with no output and every case was reported as "no result returned". The
        // case is marked instead, so it arrives with an explicit transport error and the case
        // count stays honest.
        auto ai = obj.find("\"args\"");
        auto ei = obj.find("\"expected\"");
        if (ai == string::npos || ei == string::npos) {
          out.push_back({"", "", "malformed case object: missing args or expected"});
          continue;
        }
        auto aStart = obj.find('[', ai);
        if (aStart == string::npos) {
          out.push_back({"", "", "malformed case object: args is not a list"});
          continue;
        }
        string args = obj.substr(aStart, matchBracket(obj, (int)aStart) - aStart);
        auto eStart = obj.find(':', ei);
        string exp = trim(obj.substr(eStart + 1));
        while (!exp.empty() && (exp.back() == '}' || exp.back() == ' ')) exp.pop_back();
        out.push_back({args, exp, ""});
      }
    }
  }
  return out;
}

// Parse a flat-encoded scalar or nested list into a vector<int>.
static vector<int> toIntVec(const string& s) {
  string t = trim(s);
  if (!t.empty() && t.front() == '[') t = t.substr(1, t.size() - 2);
  vector<int> out;
  for (auto& x : splitTop(t)) {
    string v = trim(x);
    if (!v.empty()) out.push_back(stoi(v));
  }
  return out;
}

static vector<vector<int>> toIntVec2(const string& s) {
  string t = trim(s);
  if (!t.empty() && t.front() == '[') t = t.substr(1, t.size() - 2);
  vector<vector<int>> out;
  for (auto& x : splitTop(t)) out.push_back(toIntVec(trim(x)));
  return out;
}

// Parse a flat-encoded list of one-character strings into a vector<char>.
//
// The wire format carries each element as a JSON string (`["5","3","."]`), because that is what
// LeetCode's own `character[]` examples look like, so the quotes have to come off before the
// first character is taken.
static vector<char> toCharVec(const string& s) {
  string t = trim(s);
  if (!t.empty() && t.front() == '[') t = t.substr(1, t.size() - 2);
  vector<char> out;
  for (auto& x : splitTop(t)) {
    string v = unquote(trim(x));
    out.push_back(v.empty() ? ' ' : v[0]);
  }
  return out;
}

static vector<vector<char>> toCharVec2(const string& s) {
  string t = trim(s);
  if (!t.empty() && t.front() == '[') t = t.substr(1, t.size() - 2);
  vector<vector<char>> out;
  for (auto& x : splitTop(t)) out.push_back(toCharVec(trim(x)));
  return out;
}

static vector<string> toStrVec(const string& s) {
  string t = trim(s);
  if (!t.empty() && t.front() == '[') t = t.substr(1, t.size() - 2);
  vector<string> out;
  for (auto& x : splitTop(t)) {
    string v = trim(x);
    if (v.size() >= 2 && v.front() == '"') v = jsonUnescape(v.substr(1, v.size() - 2));
    out.push_back(v);
  }
  return out;
}

// A quoted, escaped JSON string: the value form, not the transport form.
//
// `jsonEscape` deliberately escapes without adding quotes, because it is also used on the
// `args`/`expected` fields, which are JSON strings whose quotes the emit site already
// writes. `got`, by contrast, is emitted as a bare JSON VALUE — so a returned string needs
// its own quotes, and the quotes must not also come from the surrounding literal.
static string jsonString(const string& s) {
  string out;
  out.reserve(s.size() + 2);
  out += (char)34;
  out += jsonEscape(s);
  out += (char)34;
  return out;
}

// A JSON array of numbers. Already valid JSON on its own, so the emit site splices it bare.
static string renderInts(const vector<int>& v) {
  string o = "[";
  for (size_t i = 0; i < v.size(); i++) { if (i) o += ","; o += to_string(v[i]); }
  return o + "]";
}

// A JSON array of strings. Each element is a JSON string, not a flat-encoded token: a
// returned element containing a quote, backslash or control character must stay valid JSON,
// or the whole run is rejected as "harness emitted non-JSON".
static string renderStrings(const vector<string>& v) {
  string o = "[";
  for (size_t i = 0; i < v.size(); i++) { if (i) o += ","; o += jsonString(v[i]); }
  return o + "]";
}

static bool sameInts(vector<int> a, vector<int> b) {
  sort(a.begin(), a.end());
  sort(b.begin(), b.end());
  return a == b;
}

// Nested string arrays, for `list<list<string>>` returns (group-anagrams). Without these the
// whole problem fell into the "unsupported signature" branch in every C++ submission.
static vector<vector<string>> toStrVec2(const string& s) {
  string t = trim(s);
  if (!t.empty() && t.front() == '[') t = t.substr(1, t.size() - 2);
  vector<vector<string>> out;
  for (auto& x : splitTop(t)) out.push_back(toStrVec(trim(x)));
  return out;
}

static string renderStrVec2(const vector<vector<string>>& v) {
  string o = "[";
  for (size_t i = 0; i < v.size(); i++) { if (i) o += ","; o += renderStrings(v[i]); }
  return o + "]";
}

/**
 * Order-free comparison of two nested string arrays.
 *
 * The expected value comes from the dataset, whose own reference solution ordered its groups
 * differently from the user's — that is exactly the false rejection `verifiers.ts` exists to
 * prevent on the TypeScript side. Both the outer groups and the inner members are compared as
 * multisets, because `group-anagrams` accepts any ordering at both levels.
 */
static bool sameStrVec2(vector<vector<string>> a, vector<vector<string>> b) {
  if (a.size() != b.size()) return false;
  for (auto& g : a) sort(g.begin(), g.end());
  for (auto& g : b) sort(g.begin(), g.end());
  sort(a.begin(), a.end());
  sort(b.begin(), b.end());
  return a == b;
}
